class Solution:
    def minimumSubarrayLength(self, nums: List[int], k: int) -> int:
        count = [0] * 32  # Track bit counts
        current_or = 0
        left = 0
        result = float('inf')
        
        for right in range(len(nums)):
            num = nums[right]
            current_or |= num
            for i in range(32):
                if num & (1 << i):
                    count[i] += 1
            
            while current_or >= k and left <= right:
                result = min(result, right - left + 1)
                num_left = nums[left]
                for i in range(32):
                    if num_left & (1 << i):
                        count[i] -= 1
                        if count[i] == 0:
                            current_or &= ~(1 << i)  # Clear bit if no more contributions
                left += 1
        
        return result if result != float('inf') else -1
