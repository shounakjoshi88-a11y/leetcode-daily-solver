class Solution:
    def validSquare(self, p1: list[int], p2: list[int], p3: list[int], p4: list[int]) -> bool:
        # gather the four points into a list
        pts = [p1, p2, p3, p4]
        dists = []  # list to hold squared distances between each pair of points

        # compute squared distance for each of the 6 unique pairs
        for i in range(4):
            for j in range(i + 1, 4):
                dx = pts[i][0] - pts[j][0]
                dy = pts[i][1] - pts[j][1]
                dists.append(dx * dx + dy * dy)

        # order distances so the four smallest are sides and the largest two are diagonals
        dists.sort()

        # smallest distance must be positive (non-zero side length)
        side = dists[0]
        if side == 0:
            return False

        # verify that the first four distances (potential sides) are equal
        for i in range(1, 4):
            if dists[i] != side:
                return False

        # verify that the last two distances (potential diagonals) are equal
        if dists[4] != dists[5]:
            return False

        # ensure right angles by checking Pythagorean theorem for diagonals
        if dists[4] != 2 * side:
            return False

        return True
